cmake_minimum_required(VERSION 3.16)
project(mikrobit_firmware_core LANGUAGES CXX)
option(MIKROBIT_SANITIZERS "Enable Address/UndefinedBehaviorSanitizer on host" OFF)
add_library(mikrobit_core src/core/protocol.cpp src/core/outputs.cpp src/core/firmware.cpp)
target_include_directories(mikrobit_core PUBLIC src/core)
set_target_properties(mikrobit_core PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
  target_compile_options(mikrobit_core PRIVATE /W4 /WX /GR-)
else()
  target_compile_options(mikrobit_core PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions -fno-rtti)
  if(MIKROBIT_SANITIZERS)
    target_compile_options(mikrobit_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(mikrobit_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
include(CTest)
if(BUILD_TESTING AND NOT MIKROBIT_HOST_TESTS_STANDALONE)
  add_subdirectory(tests)
endif()
