cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(mikrobit_host_tests LANGUAGES CXX)
  enable_testing()
  set(MIKROBIT_HOST_TESTS_STANDALONE ON)
  add_subdirectory(.. firmware)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(firmware_tests firmware_tests.cpp)
target_link_libraries(firmware_tests PRIVATE mikrobit_core)
set_target_properties(firmware_tests PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
add_test(NAME firmware_core COMMAND firmware_tests)
add_executable(firmware_simulator simulator.cpp)
target_link_libraries(firmware_simulator PRIVATE mikrobit_core)
set_target_properties(firmware_simulator PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
add_test(NAME shared_fixtures COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_fixtures.py
  $<TARGET_FILE:firmware_simulator> ${CMAKE_CURRENT_SOURCE_DIR}/../../spec/contracts/microbit-webapp/fixtures/conformance.json)
